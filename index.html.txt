<script>
  // シーンのデータ定義
  let currentScene = 0;
  const scenes = [
    { 
      id: 0, 
      text: "第一章：今回の事件ではどうも１人の男が死んでいるようだ。", 
      objects: [
        { id: 'wallet', name: '財布', x: 300, y: 150, info: "身分証と…なんと２億円が入っている！" }
      ] 
    },
    { 
      id: 1, 
      text: "視点を変えた。机の上に何かある。", 
      objects: [
        { id: 'hammer', name: '金槌', x: 450, y: 200, info: "トンカチともいう。「ハンマー」として使えるかもしれない。" }
      ] 
    }
  ];
  let inventory = [];
  let isChapter1Cleared = false;
  
  // シーンの描画
  function renderScene() {
    const sceneInfo = scenes[currentScene];
    document.getElementById('scene-text').innerHTML = sceneInfo.text;
    
    let content = "";
    sceneInfo.objects.forEach(obj => {
      content += `<div class="clickable-object" style="left:${obj.x}px; top:${obj.y}px;" onclick="collectItem('${obj.id}', '${obj.name}', '${obj.info}')">${obj.name}</div>`;
    });
    document.getElementById('scene-content').innerHTML = content;
  }
  
  // 視点移動 (◀︎ ▶︎)
  function changeScene(dir) {
    currentScene += dir;
    if (currentScene < 0) currentScene = scenes.length - 1;
    if (currentScene >= scenes.length) currentScene = 0;
    renderScene();
  }
  
  // アイテム取得
  function collectItem(id, name, info) {
    // インベントリに追加
    inventory.push({id, name, info});
    // 画面から消す
    scenes[currentScene].objects = scenes[currentScene].objects.filter(obj => obj.id !== id);
    
    renderScene();
    renderInventory();
    
    // アイテム取得時のダイアログ
    alert(`${name} を手に入れた！`);
  }
  
  // アイテム欄の描画
  function renderInventory() {
    const invDiv = document.getElementById('inventory');
    invDiv.innerHTML = '';
    inventory.forEach(item => {
      invDiv.innerHTML += `<div class="item" onclick="alert('【${item.name}の詳細】\\n${item.info}')">${item.name}</div>`;
    });
  }
  
  // モーダルウィンドウの表示/非表示
  function showModal(html) {
    document.getElementById('modal-content').innerHTML = html;
    document.getElementById('modal').style.display = 'block';
  }
  function closeModal() {
    document.getElementById('modal').style.display = 'none';
  }
  
  // 「考える」ボタンの処理
  function showThink() {
    showModal(`
      <h2>Q1: 現場の違和感は？</h2>
      <p>この事件の状況から考えて、何かおかしい点はないだろうか？</p>
      <p style="color: #bdc3c7;">ヒント: 落ちていたアイテムの中身をよく思い出そう。</p>
    `);
  }
  
  // 「謎」ボタンの処理
  function showMystery() {
    showModal(`
      <h2>謎解きパート</h2>
      <p>2億円に目を奪われているようじゃ探偵失格だ。<br>財布に入っていた「もう一つの重要なもの」を入力しろ。</p>
      <input type="text" id="answer-input" placeholder="答えを入力">
      <br>
      <button onclick="checkAnswer()">解答</button>
      <h3 id="answer-feedback" style="color:#f1c40f; margin-top: 20px;"></h3>
    `);
  }
  
  // 解答チェック
  function checkAnswer() {
    const ans = document.getElementById('answer-input').value;
    const feedback = document.getElementById('answer-feedback');
    
    if (ans.includes('身分証')) {
      feedback.innerHTML = "所長「お見事！よくわかったね。<br>え！？この人が泥棒で、つまり犯人はここの住人ってこと！？<br>逃亡しているだろうね、なんとか追わねば！<br><br>ー第二章へー」";
      isChapter1Cleared = true;
    } else {
      feedback.innerHTML = "所長「うーん…違う気がするね。アイテムの詳細をもう一度タップして確認してみよう。」";
    }
  }
  
  // 初期化処理
  renderScene();
</script>
